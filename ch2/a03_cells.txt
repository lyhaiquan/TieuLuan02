def to_float(s):
    "Bóc số thực từ chuỗi kiểu Việt Nam: '86,96 triệu/m²' -> 86.96"
    if pd.isna(s):
        return np.nan
    m = re.search(r"[\d.,]+", str(s))
    if not m:
        return np.nan
    t = m.group(0).replace(".", "").replace(",", ".")   # '1.234,5' -> '1234.5'
    try:
        return float(t)
    except ValueError:
        return np.nan

def to_rooms(s):
    "'5 phòng' -> 5 ; 'nhiều hơn 10 phòng' -> 10"
    if pd.isna(s):
        return np.nan
    if "nhiều hơn" in str(s).lower():
        return 10.0
    return to_float(s)
